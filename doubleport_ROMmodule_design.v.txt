// Dual Port ROM module design

module dual_port_rom (
  input clk, //clk
  input en1, //enable for port 1
  input en2, //enable for port 2
  
  input [3:0] addr1, //address for port 1
  input [3:0] addr2, //address for port 2
  
  output reg [3:0] data1, //output data for port 1
  output reg [3:0] data2  //output data for port 2
);

  reg [3:0] mem [15:0]; //4 bit data and 16 locations

  always @(posedge clk)
    begin
      // Port 1
      if (en1)
        data1 <= mem[addr1];
      else
        data1 <= 4'bxxxx;

      // Port 2
      if (en2)
        data2 <= mem[addr2];
      else
        data2 <= 4'bxxxx;
    end

  initial
    begin
      mem[0]  = 4'b0010;
      mem[1]  = 4'b0010;
      mem[2]  = 4'b1110;
      mem[3]  = 4'b0010;
      mem[4]  = 4'b0100;
      mem[5]  = 4'b1010;
      mem[6]  = 4'b1100;
      mem[7]  = 4'b0000;
      mem[8]  = 4'b1010;
      mem[9]  = 4'b0010;
      mem[10] = 4'b1110;
      mem[11] = 4'b0010;
      mem[12] = 4'b0100;
      mem[13] = 4'b1010;
      mem[14] = 4'b1100;
      mem[15] = 4'b0000;
    end

endmodule
